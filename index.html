/* =====================================================

   SHUFFLE

===================================================== */


function shuffle(array) {


  const copy = [...array];



  for (

    let i = copy.length - 1;

    i > 0;

    i--

  ) {


    const j =

      Math.floor(

        Math.random() *

        (i + 1)

      );



    [

      copy[i],

      copy[j]

    ] = [

      copy[j],

      copy[i]

    ];


  }



  return copy;


}

function checkAnswer(

  button,

  selectedFont

) {


  if (answered) {

    return;

  }



  /* ===================================================

     DOĞRU CEVAP

  =================================================== */


  if (

    selectedFont.name ===

    currentFont.name

  ) {


    answered = true;



    button.classList.add(

      "correct"

    );



    score += 100;


    correctAnswers++;



    scoreEl.textContent =

      score;



    feedbackEl.textContent =

      `✓ Tebrikler! Doğru cevap: ${currentFont.name}`;



    feedbackEl.className =

      "feedback success";



    disableOptions();



    /*

    10. soruysa oyun biter.

    */


    if (

      round >=

      TOTAL_QUESTIONS

    ) {


      progressEl.style.width =

        "100%";



      setTimeout(

        () => endGame(false),

        1400

      );


    }


    else {


      setTimeout(

        () => {


          round++;


          newRound();


        },

        1400

      );


    }


  }



  /* ===================================================

     YANLIŞ CEVAP

  =================================================== */


  else {


    button.classList.add(

      "wrong"

    );



    button.disabled =

      true;



    lives--;


    wrongAnswers++;



    updateLives();



    feedbackEl.textContent =

      "Yanlış şüpheli. Bir kez daha incele.";



    feedbackEl.className =

      "feedback error";



    /*

    Can bittiyse oyun biter.

    */


    if (

      lives <= 0

    ) {


      answered = true;



      revealCorrectAnswer();



      feedbackEl.textContent =

        `Doğru cevap: ${currentFont.name}`;



      setTimeout(

        () => endGame(true),

        1200

      );


    }


  }


}

function endGame(

  lostByLives = false

) {


  gameScreen.classList.add(

    "hidden"

  );



  gameOverScreen.classList.add(

    "active"

  );



  /*

  Başarı oranı doğru çözülen

  soru sayısına göre hesaplanır.

  */


  const questionsPlayed =

    lostByLives

      ? round

      : TOTAL_QUESTIONS;



  const accuracy =

    questionsPlayed > 0


      ? Math.round(

          (

            correctAnswers /

            questionsPlayed

          ) * 100

        )


      : 0;



  const content =

    getResultContent(

      accuracy,

      lostByLives

    );



  finalScoreEl.textContent =

    score;



  correctResultEl.textContent =

    correctAnswers;



  wrongResultEl.textContent =

    wrongAnswers;



  accuracyResultEl.textContent =

    `%${accuracy}`;



  resultIconEl.textContent =

    content.icon;



  resultTitleEl.textContent =

    content.title;



  resultMessageEl.textContent =

    content.message;


}
